7.1.2 有序对与多个因素

Definition 7.2

A subgroup H<GH < G is called normal if each left coset of H is also a right coset of H (and vice versa).

中文陈述：H≤GH≤G 正规，指它的左陪集也都是右陪集，反之亦然。等价地，对每个 g∈Gg∈G 有 gH=HggH=Hg。记 H◁GH◁G。

来源：印刷 121 / PDF 136

在 C4×C2C₄×C₂ 中，两行与四列分别是两个自然因素的陪集；左右平移得到相同的分块，所以它们正规。一般两个因素的正规性见 7.12。A×BA×B 的元素是有序对 (a,b)(a,b)，运算逐坐标进行。

有序意味着 (a,b)(a,b) 与 (b,a)(b,a) 是不同的表达；A×BA×B 和 B×AB×A 同构但通常不是同一个集合。三个因素可写 (a,b,c)(a,b,c)，每个动作只动一个坐标。C23C₂³ 的图是立方体；C24C₂⁴ 再给每个立方体顶点增加第四个开关，共 16 个节点。第三、第四坐标用平铺副本表示时，仍要画出副本之间的连线。

7.1.3 为什么直积能简化大群

例如 D4×C7D₄×C₇ 有 56 个元素。要描述 (r2,5)(r²,5)，只需在八节点 D4D₄ 图标 r2r²，并在七节点 C7C₇ 图标 5；组合时两个坐标各自运算即可，无需把全部 56 节点挤在一张纸上。有效的简化来自乘法独立，不是只从节点数分解 56=8·756=8·7 就猜直积。D4D₄ 的八节点虽能排成 4×24×2，却有翻面造成的反向，故不是 C4×C2C₄×C₂。

7.2 半直积预览：副本可以被自同构扭转

Definition 7.4

原文节选（仅条件 (c)；完整约束见下文）

7.5 共轭：把同一个动作搬到另一个位置

共轭 ghg−1ghg⁻¹ 按时间解释为先换环境 gg，再做 hh，再恢复环境 g−1g⁻¹。代数上它是群的自同构，因为 (gxg−1)(gyg−1)=(gxg⁻¹)(gyg⁻¹)=gxyg⁻¹⁻¹。共轭保留阶；它会把一个旋转轴或交换位置搬到别处，却保留相同的动作结构。

正四面体中 a,b,c,da,b,c,d 是绕四个顶点的同方向 120° 转动；改变顶点所处位置能把它们互相共轭。按前文 A4A₄ 标签，cac−1=bcac⁻¹=b。反方向的四个转动构成另一类，在 A4A₄ 内不能把空间取向反转来连接这两类。

#!/usr/bin/env python3
"""Chapter 7: C_n semidirect C_m, with t*r*t^-1 = r^u."""
import math
import sys


def multiply(a, b, n, m, u):
    i, j = a % n, a // n
    k, ell = b % n, b // n
    return (i + pow(u, j, n) * k) % n + n * ((j + ell) % m)


def element_order(a, n, m, u):
    x = 0
    for k in range(1, n * m + 1):
        x = multiply(x, a, n, m, u)
        if x == 0:
            return k
    raise ValueError('invalid group parameters')


def main(args):
    if not args:
        args = ['4', '2', '3']
    if len(args) != 3 or any(not s.isascii() or not s.isdecimal() for s in args):
        raise ValueError('usage: products.py n m u')
    n, m, u = map(int, args)
    if not (2 <= n <= 12 and 1 <= m <= 6 and n * m <= 24 and 0 <= u < n):
        raise ValueError('require 2<=n<=12, 1<=m<=6, n*m<=24, 0<=u<n')
    if math.gcd(u, n) != 1 or pow(u, m, n) != 1:
        raise ValueError('u must be a unit modulo n with u^m=1 mod n')
    size = n * m
    print(f'n={n} m={m} u={u}; id=i+n*j represents r^i t^j')
    for a in range(size):
        print(f'{a} ({a%n},{a//n}) order={element_order(a,n,m,u)} quotient={a//n}')
    print('TABLE')
    for a in range(size):
        print(' '.join(str(multiply(a,b,n,m,u)) for b in range(size)))
    print('QUOTIENT C' + str(m))
    for j in range(m):
        print(str(j) + ': ' + ','.join(str(i + n*j) for i in range(n)))


if __name__ == '__main__':

/* Chapter 7. C_n semidirect C_m; t*r*t^-1=r^u. */
#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
static int number(const char *s, int *out) {
    int n = 0;
    if (!*s) return 0;
    for (; *s; ++s) {
        if (!isdigit((unsigned char)*s) || n > 24) return 0;
        n = 10*n + *s - '0';
    }
    *out = n; return 1;
}
static int gcd(int a, int b) {
    while (b) { int r = a%b; a=b; b=r; }
    return a;
}
static int modpow(int u, int j, int n) {
    int v = 1;
    for (int k=0; k<j; ++k) v = v*u%n;
    return v;
}
static int multiply(int a, int b, int n, int m, int u) {
    int i=a%n, j=a/n, k=b%n, ell=b/n;
    return (i+modpow(u,j,n)*k)%n + n*((j+ell)%m);
}
static int element_order(int a, int n, int m, int u) {
    int x=0;
    for (int k=1; k<=n*m; ++k) {
        x=multiply(x,a,n,m,u);
        if (!x) return k;
    }
    return 0;
}
int main(int argc, char **argv) {
    int n=4,m=2,u=3;
    if (argc != 1 && (argc != 4 || !number(argv[1],&n) ||
                     !number(argv[2],&m) || !number(argv[3],&u))) {
        fprintf(stderr,"error: expected n m u\n"); return 2;
    }
    if (n<2 || n>12 || m<1 || m>6 || n*m>24 || u<0 || u>=n ||
        gcd(u,n)!=1 || modpow(u,m,n)!=1) {
        fprintf(stderr,"error: require bounded n,m and unit u with u^m=1 mod n\n"); return 2;
    }
    int size=n*m;
    printf("n=%d m=%d u=%d; id=i+n*j represents r^i t^j\n",n,m,u);
    for (int a=0; a<size; ++a)
        printf("%d (%d,%d) order=%d quotient=%d\n",a,a%n,a/n,element_order(a,n,m,u),a/n);
    printf("TABLE\n");
    for (int a=0; a<size; ++a) {
        for (int b=0; b<size; ++b) printf("%s%d",b?" ":"",multiply(a,b,n,m,u));
        printf("\n");
    }
    printf("QUOTIENT C%d\n",m);
    for (int j=0; j<m; ++j) {
        printf("%d: ",j);
        for (int i=0; i<n; ++i) printf("%s%d",i?",":"",i+n*j);
        printf("\n");
    }
    return 0;
}
