9.1 群作用：把群元素解释成置换

拉格朗日定理只给必要条件：子群的阶必须整除群阶。A4A₄ 的阶为 12，却没有 6 阶子群，说明反向推理不成立。本章寻找可以反过来用的部分，并沿着原书的未知群 AA（|A|=200=23·52|A|=200=2³·5²）逐步增加信息。全章讨论有限群；固定点计数还要求被作用的集合有限。

Definition 9.1

原文摘录：

A group G acts on a set S if there is a homomorphism φ:G→φ : G → Perm(S).

中文陈述：群作用是同态 φ:G→Perm(S)φ:G→Perm(S)。沿用本书动作次序，写 s·g=φ(g)(s)s·g=φ(g)(s)，要求 s·e=ss·e=s，(s·a)·b=s·(ab)(s·a)·b=s·(ab)。Perm(S)(S) 的乘法是先左后右，故对应通常函数复合 φ(ab)=φ(b)∘φ(a)φ(ab)=φ(b)∘φ(a)。

来源：印刷 195 / PDF 210

作用可以丢失信息：令 V4=⟨h,v⟩V₄=⟨h,v⟩ 作用于矩形的四种配置，hh 真正水平翻转，vv 则固定所有配置。所有群关系仍成立，因而这是合法同态；但 ee 和 vv 有同一效果，核为 {e,v}{e,v}。原书用于测量物体对称性的特殊构造另要求：(1) 像恰好是手工允许的全部对称动作；(2) 映射单射。一般群作用不要求这两项。

#!/usr/bin/env python3
"""Explicit Sylow subgroups in S_n, only for 2 <= n <= 5."""
import itertools
import sys


def compose(a, b):
    """Execute a first, then b, as in the book."""
    return tuple(b[x] for x in a)


def model(n, p):
    if n not in range(2, 6) or p not in (2, 3, 5) or p > n:
        raise ValueError("2 <= n <= 5; p must be a prime <= n")
    perms = list(itertools.permutations(range(n)))
    index = {x: i for i, x in enumerate(perms)}
    table = [[index[compose(a, b)] for b in perms] for a in perms]
    inv = [row.index(0) for row in table]
    cycle = list(range(n))
    length = 4 if p == 2 and n >= 4 else p
    for i in range(length):
        cycle[i] = (i + 1) % length
    gens = [index[tuple(cycle)]]
    if length == 4:
        flip = list(range(n))
        flip[0], flip[2] = flip[2], flip[0]
        gens.append(index[tuple(flip)])
    seed = {0}
    queue = [0]
    for a in queue:
        for b in gens:
            x = table[a][b]
            if x not in seed:
                seed.add(x)
                queue.append(x)
    subgroups = sorted({tuple(sorted(table[table[g][h]][inv[g]] for h in seed))
                        for g in range(len(perms))})
    normalizers = []
    for h in subgroups:
        normalizers.append(sum(tuple(sorted(table[table[g][x]][inv[g]] for x in h)) == h
                               for g in range(len(perms))))
    return perms, subgroups, normalizers


def output(n, p):
    perms, groups, normalizers = model(n, p)
    lines = [f"S{n} order={len(perms)} p={p} sylow_order={len(groups[0])}"]
    for i, x in enumerate(perms):
        lines.append(f"perm {i}: " + ",".join(str(y + 1) for y in x))
    for h, norm in zip(groups, normalizers):
        lines.append("H={" + ",".join(map(str, h)) + f"}} normalizer={norm}")
    lines.append(f"sylow_count={len(groups)}")
    lines.append(f"point_orbit={n} point_stabilizer={sum(x[0] == 0 for x in perms)}")
    return "\n".join(lines) + "\n"


def main(args):
    try:
        if not args:
            n, p = 4, 2
        elif len(args) == 2 and all(len(x) == 1 and x in "2345" for x in args):
            n, p = map(int, args)
        else:
            raise ValueError("usage: sylow [n p]")
        result = output(n, p)
    except ValueError as error:
        print(error, file=sys.stderr)
        return 2
    print(result, end="")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

#include <stdio.h>
#include <string.h>

#define MAX 120
static int n, count;
static int perms[MAX][5], table[MAX][MAX], inverse[MAX];
static int groups[MAX][MAX], sizes[MAX], group_count;

static void enumerate(int depth, int *a, int *used) {
    if (depth == n) {
        for (int i = 0; i < n; ++i) perms[count][i] = a[i];
        ++count;
        return;
    }
    for (int i = 0; i < n; ++i) if (!used[i]) {
        used[i] = 1; a[depth] = i;
        enumerate(depth + 1, a, used);
        used[i] = 0;
    }
}

static int lookup(const int *a) {
    for (int i = 0; i < count; ++i) {
        int equal = 1;
        for (int j = 0; j < n; ++j) if (perms[i][j] != a[j]) equal = 0;
        if (equal) return i;
    }
    return -1;
}

static int compare(const int *a, const int *b, int size) {
    for (int i = 0; i < size; ++i) {
        if (a[i] < b[i]) return -1;
        if (a[i] > b[i]) return 1;
    }
    return 0;
}

static void conjugate(const int *h, int size, int g, int *out) {
    int seen[MAX] = {0}, k = 0;
    for (int i = 0; i < size; ++i) seen[table[table[g][h[i]]][inverse[g]]] = 1;
    for (int i = 0; i < count; ++i) if (seen[i]) out[k++] = i;
}
