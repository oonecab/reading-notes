5.1 循环群：重复一个动作会得到什么

本章的五类是循环群、阿贝尔群、二面体群、对称群、交错群。它们不是互不相交的五个盒子：循环群一定阿贝尔，S3S₃ 与 D3D₃ 又是同一个抽象群。本章先建立可计算的例子，再用 Cayley 定理说明任何群都能忠实地表现为置换。

5.1.1 只会旋转的物体

有 nn 片同向弯曲叶片的理想风车，每转 360°/n/n 回到同类位置，记这一步为 rr。允许的动作是 e,r,…,rn−1e,r,…,rⁿ⁻¹；rn=erⁿ=e。叶片的弯曲方向排除了反射。两个动作 rirⁱ、rjrʲ 合起来就是 ri+jrⁱ⁺ʲ，指数相差 nn 的动作相同。这给出 nn 阶循环群 CnCₙ；nn 是动作总数，不是生成元个数。

5.1.4 引入循环图

循环图把最大循环子群画成无向环，各环中的同一元素只画一次；二阶循环以一条边表示。它强调幂与阶，不指定一套生成元。它一般不能独自恢复任意两个元素的乘积；画得像一个多边形也不能忽略边是否带方向、颜色和生成元含义。

5.2 阿贝尔群：先后顺序能否交换

5.2.1 两条路径为什么必须到同一点

若任意 a,b∈Ga,b∈G 都有 ab=baab=ba，群就是阿贝尔群。图上从任意起点 xx 先走 aa 再走 bb，与先走 bb 再走 aa，终点分别是 xabx a b、xbax b a。相等的不是两段曲线的形状，而是最终节点。只需验证生成元两两交换：在任意两个生成元字词间，反复交换相邻字母便可把整块字词换位；逆元也继承交换关系。

“只需检查生成元”不等于“找到一对交换的元素即可”。每个群都有 ee 与其他元素交换；非交换群也可能有很大的交换子群。

5.2.2 表的对称与直积预告

5.2.3 循环之间怎样相交

C2×C2×C2C₂×C₂×C₂ 的七个非单位元都有阶 2，所以循环图是从 ee 发出的七条边。C2×C4C₂×C₄ 有两个四阶循环，共享 ee 与 (0,2)，另有两个二阶循环。一般在 Cp×CpCₚ×Cₚ（pp 为素数）中，每个非零元素的阶都是 pp：pp 次相加为零，而更小的正整数倍不会使非零坐标归零。

非平凡循环数=(p2−1)/(p−1)=p+1。非平凡循环数 = (p²−1)/(p−1) = p+1。

分子数出所有非零元素；一个循环贡献 p−1p−1 个；若两个循环共享非零元素，这个元素已生成各自的整个循环，所以它们其实相同。由此可除而且不会重复计数。完整证明和循环图见习题 5.28、8.23。

5.3 二面体群：翻面会倒转旋转方向

5.3.1 正多边形的全部动作

对正 nn 边形，rr 旋转一格，ff 沿某条对称轴翻面。一个刚体对称先决定顶点 0 去哪里（nn 种），再决定环绕方向（2 种），所以 n≥3n≥3 时共有 2n2n 个动作。保持方向的是 rirⁱ，反转方向的是 rifrⁱf。本书写 DnDₙ 表示这个 2n2n 阶群；有的教材会把同一个群记作 D2nD₂ₙ。这里也按原书把 D1=C2D₁=C₂、D2=V4D₂=V₄ 纳入代数定义。

5.4.5 如何使用这五类群

判断同构时先看阶，再看是否交换、元素阶和循环如何相交；这些量必须在同构下保持。寻找反例时，CnCₙ 提供单循环，V4V₄ 提供非循环的交换群，D3D₃ 提供最小非交换群，Q4Q₄ 提供阶为 8 却不属于本章五类的例子。后续的子群、商群和同态都将在这份例子库上展开。

#!/usr/bin/env python3
"""Chapter 5: permutations of 1..n, with p*q meaning p THEN q."""
import math
import sys


def parse(text):
    fields = text.split(',')
    if not 1 <= len(fields) <= 8 or any(not x.isascii() or not x.isdecimal() for x in fields):
        raise ValueError('use 1..8 comma-separated positive integers')
    p = tuple(int(x) - 1 for x in fields)
    if sorted(p) != list(range(len(p))):
        raise ValueError('each value 1..n must occur exactly once')
    return p


def compose(p, q):
    """Array entries are destinations; visit p[x], then q[p[x]]."""
    return tuple(q[x] for x in p)


def inverse(p):
    q = [0] * len(p)
    for source, destination in enumerate(p):
        q[destination] = source
    return tuple(q)


def cycles(p):
    seen = set()
    result = []
    for start in range(len(p)):
        if start in seen:
            continue
        cycle = []
        x = start
        while x not in seen:
            seen.add(x)
            cycle.append(x)
            x = p[x]
        result.append(cycle)
    return result


def describe(label, p):
    parts = cycles(p)
    text = ''.join('(' + ' '.join(str(x + 1) for x in part) + ')'
                   for part in parts if len(part) > 1) or '()'
    order = math.lcm(*(len(part) for part in parts))
    inversions = sum(p[i] > p[j] for i in range(len(p)) for j in range(i + 1, len(p)))
    print(label, ','.join(str(x + 1) for x in p), text,
          'order=' + str(order), 'sign=' + str(-1 if inversions % 2 else 1))


def main(args):
    if not args:
        args = ['2,3,4,1', '2,1,3,4']
    if len(args) != 2:
        raise ValueError('usage: permutations.py p q; e.g. 2,3,4,1 2,1,3,4')
    p, q = map(parse, args)
    if len(p) != len(q):
        raise ValueError('p and q must have the same degree')
    for label, permutation in [('p', p), ('q', q), ('p*q', compose(p, q)),
                               ('q*p', compose(q, p)), ('p^-1', inverse(p)),
                               ('p^2', compose(p, p))]:
        describe(label, permutation)


if __name__ == '__main__':
    try:
        main(sys.argv[1:])
    except ValueError as error:
        print('error:', error, file=sys.stderr)
        sys.exit(2)

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#define MAX_N 8

typedef struct { int n; int to[MAX_N]; } Perm;

static int parse(const char *text, Perm *p) {
    const char *s = text;
    p->n = 0;
    while (*s) {
        if (p->n == MAX_N || !isdigit((unsigned char)*s)) return 0;
        int value = 0;
        while (isdigit((unsigned char)*s)) {
            value = value * 10 + (*s++ - '0');
            if (value > MAX_N) return 0;
        }
        if (value < 1) return 0;
        p->to[p->n++] = value - 1;
        if (!*s) break;
        if (*s++ != ',' || !*s) return 0;
    }
    if (p->n == 0) return 0;
    int seen[MAX_N] = {0};
    for (int i = 0; i < p->n; ++i) {
        int v = p->to[i];
        if (v >= p->n || seen[v]) return 0;
        seen[v] = 1;
    }
    return 1;
}

static Perm compose(Perm p, Perm q) {
    Perm r = p;
    for (int i = 0; i < p.n; ++i) r.to[i] = q.to[p.to[i]];
    return r;
}
static Perm inverse(Perm p) {
    Perm q = p;
    for (int i = 0; i < p.n; ++i) q.to[p.to[i]] = i;
    return q;
}
static int gcd(int a, int b) {
    while (b) { int r = a % b; a = b; b = r; }
    return a;
}
static void describe(const char *label, Perm p) {
    printf("%s ", label);
    for (int i = 0; i < p.n; ++i) printf("%s%d", i ? "," : "", p.to[i] + 1);
    printf(" ");
    int seen[MAX_N] = {0}, order = 1, printed = 0, inversions = 0;
    for (int i = 0; i < p.n; ++i) {
        if (seen[i]) continue;
        int part[MAX_N], size = 0, x = i;
        while (!seen[x]) { seen[x] = 1; part[size++] = x; x = p.to[x]; }
        order = order / gcd(order, size) * size;
        if (size > 1) {
            printf("(");
            for (int j = 0; j < size; ++j) printf("%s%d", j ? " " : "", part[j] + 1);
            printf(")");
            printed = 1;
        }
    }
    if (!printed) printf("()");
    for (int i = 0; i < p.n; ++i)
        for (int j = i + 1; j < p.n; ++j) inversions += p.to[i] > p.to[j];
    printf(" order=%d sign=%d\n", order, inversions % 2 ? -1 : 1);
}

    const char *a = "2,3,4,1", *b = "2,1,3,4";
